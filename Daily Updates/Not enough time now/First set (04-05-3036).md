#### Using a Tree-map like structure to solve the Calendar II problem

```python
class MyCalendarTwo:
    def __init__(self):
        # Keep track of single bookings and double bookings separately
        self.single_bookings = []
        self.double_bookings = []
    
    def book(self, startTime: int, endTime: int) -> bool:
        # Check if it creates triple booking with existing double bookings
        for s, e in self.double_bookings:
            if max(startTime, s) < min(endTime, e):
                return False
        
        # Add to double bookings where it overlaps with single bookings
        for s, e in self.single_bookings:
            overlap_start = max(startTime, s)
            overlap_end = min(endTime, e)
            if overlap_start < overlap_end:
                self.double_bookings.append((overlap_start, overlap_end))
        
        self.single_bookings.append((startTime, endTime))
        return True
```
