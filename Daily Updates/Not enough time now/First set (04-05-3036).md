#### Using a Tree-map like structure to solve the Calendar II problem

```python
class MyCalendarTwo:
    def __init__(self):
        # Keep track of single bookings and double bookings separately
        self.single_bookings = []
        self.double_bookings = []
    
    def book(self, startTime: int, endTime: int) -> bool:
        # Check if it creates triple booking with existing double bookings
        for s, e in self.double_bookings:
            if max(startTime, s) < min(endTime, e):
                return False
        
        # Add to double bookings where it overlaps with single bookings
        for s, e in self.single_bookings:
            overlap_start = max(startTime, s)
            overlap_end = min(endTime, e)
            if overlap_start < overlap_end:
                self.double_bookings.append((overlap_start, overlap_end))
        
        self.single_bookings.append((startTime, endTime))
        return True
```

Fenwick Tree approach to problem: shifting-letters-ii
```python
class SimpleBIT:
    def __init__ (self, n): self.arr = [0 for i in range(n + 1)]
    def update (self, idx, val):
        while (idx <= (len(self.arr) - 1)):
            self.arr[idx] += val
            idx += (idx & -idx)
    def sum (self, idx):
        ans = 0
        while (idx > 0):
            ans += self.arr[idx]
            idx -= (idx & -idx)
        return ans

class Solution:
    def shiftingLetters (self, s: str, shifts: List[List[int]]) -> str:
        bit = SimpleBIT(len(s))
        for start, end, direc in shifts:
            bit.update(start + 1, 1 if direc == 1 else -1)
            bit.update(end + 2, -1 if direc == 1 else 1)
        ans = []
        for i in range(len(s)):
            ch_idx = ord(s[i]) - ord('a')
            new_ch_idx = ((ch_idx + bit.sum(i + 1)) % 26) + ord('a')
            ans.append(chr(new_ch_idx))
        return "".join(ans)
```
